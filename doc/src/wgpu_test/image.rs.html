10#[cfg(not(any(target_arch = "wasm32", miri)))]
11async fn read_png(path: impl AsRef<Path>, width: u32, height: u32) -> Option<Vec<u8>> {
12    let data = match std::fs::read(&path) {
13        Ok(f) => f,
14        Err(e) => {
15            log::warn!(
16                "image comparison invalid: file io error when comparing {}: {}",
17                path.as_ref().display(),
18                e
19            );
20            return None;
21        }
22    };
23    let decoder = png::Decoder::new(std::io::Cursor::new(data));
24    let mut reader = decoder.read_info().ok()?;
25
26    let buffer_len = reader
27        .output_buffer_size()
28        .expect("output buffer would not fit in memory");
29    let mut buffer = vec![0; buffer_len];
30    let info = reader.next_frame(&mut buffer).ok()?;
31    if info.width != width {
32        log::warn!("image comparison invalid: size mismatch");
33        return None;
34    }
35    if info.height != height {
36        log::warn!("image comparison invalid: size mismatch");
37        return None;
38    }
39    if info.color_type != png::ColorType::Rgba {
40        log::warn!("image comparison invalid: color type mismatch");
41        return None;
42    }
43    if info.bit_depth != png::BitDepth::Eight {
44        log::warn!("image comparison invalid: bit depth mismatch");
45        return None;
46    }
47
48    Some(buffer)
49}

51#[cfg(not(any(target_arch = "wasm32", miri)))]
52async fn write_png(
53    path: impl AsRef<Path>,
54    width: u32,
55    height: u32,
56    data: &[u8],
57    compression: png::Compression,
58) {
59    let file = std::io::BufWriter::new(std::fs::File::create(path).unwrap());
60
61    let mut encoder = png::Encoder::new(file, width, height);
62    encoder.set_color(png::ColorType::Rgba);
63    encoder.set_depth(png::BitDepth::Eight);
64    encoder.set_compression(compression);
65    let mut writer = encoder.write_header().unwrap();
66
67    writer.write_image_data(data).unwrap();
68}

70#[cfg_attr(any(target_arch = "wasm32", miri), allow(unused))]
71fn add_alpha(input: &[u8]) -> Vec<u8> {
72    input
73        .chunks_exact(3)
74        .flat_map(|chunk| [chunk[0], chunk[1], chunk[2], 255])
75        .collect()
76}

79fn remove_alpha(input: &[u8]) -> Vec<u8> {
80    input
81        .chunks_exact(4)

87#[cfg(not(any(target_arch = "wasm32", miri)))]
88fn print_flip(pool: &mut nv_flip::FlipPool) {
89    println!("\tMean: {:.6}", pool.mean());
90    println!("\tMin Value: {:.6}", pool.min_value());
91    for percentile in [25, 50, 75, 95, 99] {
92        println!(
93            "\t      {percentile}%: {:.6}",
94            pool.get_percentile(percentile as f32 / 100.0, true)
95        );
96    }
97    println!("\tMax Value: {:.6}", pool.max_value());
98}

100/// The FLIP library generates a per-pixel error map where 0.0 represents "no error"
101/// and 1.0 represents "maximum error" between the images. This is then put into
102/// a weighted-histogram, which we query to determine if the errors between
103/// the test and reference image is high enough to count as "different".
104///
105/// Error thresholds will be different for every test, but good initial values
106/// to look at are in the [0.01, 0.1] range. The larger the area that might have
107/// inherent variance, the larger this base value is. Using a high percentile comparison
108/// (e.g. 95% or 99%) is good for images that are likely to have a lot of error
109/// in a small area when they fail.
110#[derive(Debug, Clone, Copy)]
111pub enum ComparisonType {
112    /// If the mean error is greater than the given value, the test will fail.
113    Mean(f32),
114    /// If the given percentile is greater than the given value, the test will fail.
115    ///
116    /// The percentile is given in the range [0, 1].
117    Percentile { percentile: f32, threshold: f32 },
118}
119
120impl ComparisonType {
121    #[cfg(not(any(target_arch = "wasm32", miri)))]
122    fn check(&self, pool: &mut nv_flip::FlipPool) -> bool {
123        match *self {
124            ComparisonType::Mean(v) => {
125                let mean = pool.mean();
126                let within = mean <= v;
127                println!(
128                    "\tExpected Mean ({:.6}) to be under expected maximum ({}): {}",
129                    mean,
130                    v,
131                    if within { "PASS" } else { "FAIL" }
132                );
133                within
134            }
135            ComparisonType::Percentile {
136                percentile: p,
137                threshold: v,
138            } => {
139                let percentile = pool.get_percentile(p, true);
140                let within = percentile <= v;
141                println!(
142                    "\tExpected {}% ({:.6}) to be under expected maximum ({}): {}",
143                    p * 100.0,
144                    percentile,
145                    v,
146                    if within { "PASS" } else { "FAIL" }
147                );
148                within
149            }
150        }
151    }
152}

154#[cfg(not(any(target_arch = "wasm32", miri)))]
155pub async fn compare_image_output(
156    path: impl AsRef<Path> + AsRef<OsStr>,
157    adapter_info: &wgpu::AdapterInfo,
158    width: u32,
159    height: u32,
160    test_with_alpha: &[u8],
161    checks: &[ComparisonType],
162) {
163    use std::{ffi::OsString, str::FromStr};
164
165    let reference_path = Path::new(&path);
166    let reference_with_alpha = read_png(&path, width, height).await;
167
168    let reference = match reference_with_alpha {
169        Some(v) => remove_alpha(&v),
170        None => {
171            write_png(
172                &path,
173                width,
174                height,
175                test_with_alpha,
176                png::Compression::High,
177            )
178            .await;
179            return;
180        }
181    };
182    let test = remove_alpha(test_with_alpha);
183
184    assert_eq!(reference.len(), test.len());
185
186    let file_stem = reference_path.file_stem().unwrap().to_string_lossy();
187    let renderer = format!(
188        "{}-{}-{}",
189        adapter_info.backend,
190        sanitize_for_path(&adapter_info.name),
191        sanitize_for_path(&adapter_info.driver)
192    );
193    // Determine the paths to write out the various intermediate files
194    let actual_path = Path::new(&path)
195        .with_file_name(OsString::from_str(&format!("{file_stem}-{renderer}-actual.png")).unwrap());
196    let difference_path = Path::new(&path).with_file_name(
197        OsString::from_str(&format!("{file_stem}-{renderer}-difference.png",)).unwrap(),
198    );
199
200    let mut all_passed;
201    let magma_image_with_alpha;
202    {
203        let reference_flip = nv_flip::FlipImageRgb8::with_data(width, height, &reference);
204        let test_flip = nv_flip::FlipImageRgb8::with_data(width, height, &test);
205
206        let error_map_flip = nv_flip::flip(
207            reference_flip,
208            test_flip,
209            nv_flip::DEFAULT_PIXELS_PER_DEGREE,
210        );
211        let mut pool = nv_flip::FlipPool::from_image(&error_map_flip);
212
213        println!(
214            "Starting image comparison test with reference image \"{}\"",
215            reference_path.display()
216        );
217
218        print_flip(&mut pool);
219
220        // If there are no checks, we want to fail the test.
221        all_passed = !checks.is_empty();
222        // We always iterate all of these, as the call to check prints
223        for check in checks {
224            all_passed &= check.check(&mut pool);
225        }
226
227        // Convert the error values to a false color representation
228        let magma_image = error_map_flip
229            .apply_color_lut(&nv_flip::magma_lut())
230            .to_vec();
231        magma_image_with_alpha = add_alpha(&magma_image);
232    }
233
234    write_png(
235        actual_path,
236        width,
237        height,
238        test_with_alpha,
239        png::Compression::Fast,
240    )
241    .await;
242    write_png(
243        &difference_path,
244        width,
245        height,
246        &magma_image_with_alpha,
247        png::Compression::Fast,
248    )
249    .await;
250
251    if !all_passed {
252        panic!("Image data mismatch: {}", difference_path.display())
253    }
254}

278/// Value that readback buffers are filled with before use, so that a copy that fails to
279/// happen is not mistaken for zeroed memory.
280const POISON: u8 = 255;
281
282/// Layout of one mip level of one aspect of a texture within a readback buffer.
283///
284/// The array layers (for `D2` textures) or depth slices (for `D3`) of a mip level are laid
285/// out consecutively within its region, each `bytes_per_row * rows_per_image` bytes long.
286#[derive(Clone, Copy, Debug)]
287struct MipLayout {
288    /// Byte offset of this mip level's region within the buffer. A multiple of
289    /// [`COPY_BYTES_PER_ROW_ALIGNMENT`], which satisfies every alignment requirement that
290    /// applies: copy offsets, storage binding offsets, and buffer mapping.
291    offset: u64,
292    /// Row stride within the buffer. Aligned to [`COPY_BYTES_PER_ROW_ALIGNMENT`], which
293    /// `copy_texture_to_buffer` requires, except on the compute shader path, whose shader
294    /// writes tightly packed rows.
295    bytes_per_row: u32,
296    /// Bytes of texel data in each row, without the padding included in `bytes_per_row`.
297    unpadded_bytes_per_row: u32,
298    /// Number of block rows in each array layer or depth slice.
299    rows_per_image: u32,
300    /// Physical (block aligned) extent of this mip level.
301    size: Extent3d,
302    /// Bytes reserved for this mip level, including the padding that keeps the offset of
303    /// the next mip level aligned.
304    region_size: u64,
305}
306
307impl MipLayout {
308    /// Bytes of texel data in one array layer or depth slice, excluding row padding.
309    fn unpadded_subresource_size(&self) -> usize {
310        self.unpadded_bytes_per_row as usize * self.rows_per_image as usize
311    }
312
313    /// Bytes of texel data in this mip level, excluding all padding.
314    fn unpadded_size(&self) -> usize {
315        self.unpadded_subresource_size() * self.size.depth_or_array_layers as usize
316    }
317}

319/// Whether the readable aspect of `format` has to be read with a compute shader because it
320/// cannot be the source of a `copy_texture_to_buffer`.
321fn needs_compute_readback(format: TextureFormat) -> bool {
322    matches!(
323        format,
324        TextureFormat::Depth24Plus | TextureFormat::Depth24PlusStencil8
325    )
326}

328/// Compute the layout of every mip level of one aspect of `texture`.
329///
330/// When `dense_rows` is `true`, computes a layout with tightly packed rows for
331/// the compute shader path. When false, computes a layout with padding to
332/// `COPY_BYTES_PER_ROW_ALIGNMENT` for `copy_texture_to_buffer`.
333fn mip_layouts(
334    texture: &Texture,
335    aspect: Option<TextureAspect>,
336    dense_rows: bool,
337) -> Vec<MipLayout> {
338    let format = texture.format();
339    let (block_width, block_height) = format.block_dimensions();
340    // `block_copy_size` returns `None` for the aspects that are read with a compute
341    // shader, which writes one `f32` or `u32` per texel.
342    let block_size = format.block_copy_size(aspect).unwrap_or(4);
343
344    let mut offset = 0;
345    (0..texture.mip_level_count())
346        .map(|mip_level| {
347            // The physical size is what a copy of a whole mip level must use, and it is
348            // what the buffer has to hold: for a compressed format, the last row or
349            // column of blocks extends past the logical size.
350            let size = texture
351                .size()
352                .mip_level_size(mip_level, texture.dimension())
353                .physical_size(format);
354            let unpadded_bytes_per_row = (size.width / block_width) * block_size;
355            let bytes_per_row = if dense_rows {
356                unpadded_bytes_per_row
357            } else {
358                align_to(unpadded_bytes_per_row, COPY_BYTES_PER_ROW_ALIGNMENT)
359            };
360            let rows_per_image = size.height / block_height;
361            let layout = MipLayout {
362                offset,
363                bytes_per_row,
364                unpadded_bytes_per_row,
365                rows_per_image,
366                size,
367                region_size: align_to(
368                    u64::from(bytes_per_row)
369                        * u64::from(rows_per_image)
370                        * u64::from(size.depth_or_array_layers),
371                    u64::from(COPY_BYTES_PER_ROW_ALIGNMENT),
372                ),
373            };
374            offset += layout.region_size;
375            layout
376        })
377        .collect()
378}

380fn create_readback_buffer(device: &Device, label: &str, layouts: &[MipLayout]) -> Buffer {
381    let size = layouts.last().map_or(0, |l| l.offset + l.region_size);
382    device.create_buffer_init(&util::BufferInitDescriptor {
383        label: Some(label),
384        usage: BufferUsages::MAP_READ | BufferUsages::COPY_DST,
385        contents: &vec![POISON; size as usize],
386    })
387}

389fn copy_via_compute(
390    device: &Device,
391    encoder: &mut CommandEncoder,
392    texture: &Texture,
393    buffer: &Buffer,
394    layouts: &[MipLayout],
395    aspect: TextureAspect,
396) {
397    assert_eq!(
398        texture.dimension(),
399        TextureDimension::D2,
400        "the compute shader readback path binds the texture as `D2Array`"
401    );
402
403    let bgl = device.create_bind_group_layout(&BindGroupLayoutDescriptor {
404        label: None,
405        entries: &[
406            BindGroupLayoutEntry {
407                binding: 0,
408                visibility: ShaderStages::COMPUTE,
409                ty: BindingType::Texture {
410                    sample_type: match aspect {
411                        TextureAspect::DepthOnly => TextureSampleType::Float { filterable: false },
412                        TextureAspect::StencilOnly => TextureSampleType::Uint,
413                        _ => unreachable!(),
414                    },
415                    view_dimension: TextureViewDimension::D2Array,
416                    multisampled: false,
417                },
418                count: None,
419            },
420            BindGroupLayoutEntry {
421                binding: 1,
422                visibility: ShaderStages::COMPUTE,
423                ty: BindingType::Buffer {
424                    ty: BufferBindingType::Storage { read_only: false },
425                    has_dynamic_offset: false,
426                    min_binding_size: None,
427                },
428                count: None,
429            },
430        ],
431    });
432
433    // The shader writes one texel at a time, so anything it fails to cover keeps the
434    // poison value rather than reading back as zero.
435    let output_buffer = device.create_buffer_init(&util::BufferInitDescriptor {
436        label: Some("output buffer"),
437        usage: BufferUsages::COPY_SRC | BufferUsages::STORAGE,
438        contents: &vec![POISON; buffer.size() as usize],
439    });
440
441    // A view of a single mip level rebases mip level 0 onto it, so the shader reads the
442    // right level without knowing which one it is.
443    let bind_groups: Vec<BindGroup> = layouts
444        .iter()
445        .enumerate()
446        .map(|(mip_level, layout)| {
447            let view = texture.create_view(&TextureViewDescriptor {
448                aspect,
449                dimension: Some(TextureViewDimension::D2Array),
450                base_mip_level: mip_level as u32,
451                mip_level_count: Some(1),
452                ..Default::default()
453            });
454            device.create_bind_group(&BindGroupDescriptor {
455                label: None,
456                layout: &bgl,
457                entries: &[
458                    BindGroupEntry {
459                        binding: 0,
460                        resource: BindingResource::TextureView(&view),
461                    },
462                    BindGroupEntry {
463                        binding: 1,
464                        resource: BindingResource::Buffer(BufferBinding {
465                            buffer: &output_buffer,
466                            offset: layout.offset,
467                            size: BufferSize::new(layout.region_size),
468                        }),
469                    },
470                ],
471            })
472        })
473        .collect();
474
475    let pll = device.create_pipeline_layout(&PipelineLayoutDescriptor {
476        label: None,
477        bind_group_layouts: &[Some(&bgl)],
478        immediate_size: 0,
479    });
480
481    let source = String::from(include_str!("copy_texture_to_buffer.wgsl"));
482
483    let processed_source = source.replace(
484        "{{type}}",
485        match aspect {
486            TextureAspect::DepthOnly => "f32",
487            TextureAspect::StencilOnly => "u32",
488            _ => unreachable!(),
489        },
490    );
491
492    let sm = device.create_shader_module(ShaderModuleDescriptor {
493        label: Some("shader copy_texture_to_buffer.wgsl"),
494        source: ShaderSource::Wgsl(Cow::Borrowed(&processed_source)),
495    });
496
497    let pipeline_copy = device.create_compute_pipeline(&ComputePipelineDescriptor {
498        label: Some("pipeline read"),
499        layout: Some(&pll),
500        module: &sm,
501        entry_point: Some("copy_texture_to_buffer"),
502        compilation_options: Default::default(),
503        cache: None,
504    });
505
506    {
507        let mut pass = encoder.begin_compute_pass(&ComputePassDescriptor::default());
508
509        pass.set_pipeline(&pipeline_copy);
510        for bind_group in &bind_groups {
511            pass.set_bind_group(0, bind_group, &[]);
512            pass.dispatch_workgroups(1, 1, 1);
513        }
514    }
515
516    encoder.copy_buffer_to_buffer(&output_buffer, 0, buffer, 0, output_buffer.size());
517}

519fn copy_texture_to_buffer_with_aspect(
520    encoder: &mut CommandEncoder,
521    texture: &Texture,
522    buffer: &Buffer,
523    layouts: &[MipLayout],
524    aspect: TextureAspect,

547/// Buffers holding a readback of every mip level and every array layer or depth slice of a
548/// texture.
549pub struct ReadbackBuffers {
550    /// format of the texture this was created for
551    texture_format: TextureFormat,
552    /// buffer for color or depth aspects
553    buffer: Buffer,
554    /// layout of `buffer`, one entry per mip level
555    layouts: Vec<MipLayout>,
556    /// buffer for stencil aspect
557    buffer_stencil: Option<Buffer>,
558    /// layout of `buffer_stencil`, empty if there is none
559    stencil_layouts: Vec<MipLayout>,
560}
561
562impl ReadbackBuffers {
563    pub fn new(device: &Device, texture: &Texture) -> Self {
564        let texture_format = texture.format();
565        let combined = texture_format.is_combined_depth_stencil_format();
566
567        // When using the compute shader path (`depth24plus` and
568        // `depth24plus-stencil8`), rows are packed tightly instead of padded to
569        // `COPY_BYTES_PER_ROW_ALIGNMENT`.
570        let layouts = mip_layouts(
571            texture,
572            combined.then_some(TextureAspect::DepthOnly),
573            needs_compute_readback(texture_format),
574        );
575        let buffer = create_readback_buffer(device, "Texture Readback", &layouts);
576
577        let (buffer_stencil, stencil_layouts) = if combined {
578            let layouts = mip_layouts(texture, Some(TextureAspect::StencilOnly), false);
579            let buffer =
580                create_readback_buffer(device, "Texture Stencil-Aspect Readback", &layouts);
581            (Some(buffer), layouts)
582        } else {
583            (None, Vec::new())
584        };
585
586        ReadbackBuffers {
587            texture_format,
588            buffer,
589            layouts,
590            buffer_stencil,
591            stencil_layouts,
592        }
593    }
594
595    pub fn copy_from(&self, device: &Device, encoder: &mut CommandEncoder, texture: &Texture) {
596        assert_eq!(
597            (texture.format(), texture.mip_level_count() as usize),
598            (self.texture_format, self.layouts.len()),
599            "texture does not match the one this `ReadbackBuffers` was created for"
600        );
601
602        if needs_compute_readback(self.texture_format) {
603            copy_via_compute(
604                device,
605                encoder,
606                texture,
607                &self.buffer,
608                &self.layouts,
609                TextureAspect::DepthOnly,
610            );
611        } else {
612            let aspect = if self.buffer_stencil.is_some() {
613                TextureAspect::DepthOnly
614            } else {
615                TextureAspect::All
616            };
617            copy_texture_to_buffer_with_aspect(
618                encoder,
619                texture,
620                &self.buffer,
621                &self.layouts,
622                aspect,
623            );
624        }
625
626        if let Some(buffer_stencil) = &self.buffer_stencil {
627            copy_texture_to_buffer_with_aspect(
628                encoder,
629                texture,
630                buffer_stencil,
631                &self.stencil_layouts,
632                TextureAspect::StencilOnly,
633            );
634        }
635    }
636
637    /// Map `buffer` and return its texel data, tightly packed, with the row padding that
638    /// `layouts` describes removed. Mip level 0 comes first.
639    ///
640    /// The caller is responsible for unmapping `buffer`.
641    async fn retrieve_buffer(
642        &self,
643        ctx: &TestingContext,
644        buffer: &Buffer,
645        layouts: &[MipLayout],
646    ) -> Vec<u8> {
647        let buffer_slice = buffer.slice(..);
648        buffer_slice.map_async(MapMode::Read, |_| ());
649        ctx.async_poll(PollType::wait_indefinitely()).await.unwrap();
650        let data: BufferView = buffer_slice.get_mapped_range().unwrap();
651
652        let mut result = Vec::with_capacity(layouts.iter().map(MipLayout::unpadded_size).sum());
653        for layout in layouts {
654            let start = layout.offset as usize;
655            let rows = layout.rows_per_image as usize * layout.size.depth_or_array_layers as usize;
656            let region = &data[start..start + layout.bytes_per_row as usize * rows];
657            for row in region.chunks_exact(layout.bytes_per_row as usize) {
658                result.extend_from_slice(&row[..layout.unpadded_bytes_per_row as usize]);
659            }
660        }
661        result
662    }
663
664    /// Read back one aspect of the texture.
665    ///
666    /// Reads the stencil buffer if `aspect` is [`TextureAspect::StencilOnly`] and the
667    /// texture has a separate stencil aspect, and the color or depth buffer otherwise.
668    pub async fn retrieve(&self, ctx: &TestingContext, aspect: TextureAspect) -> ReadbackData {
669        let (buffer, layouts) = match (aspect, &self.buffer_stencil) {
670            (TextureAspect::StencilOnly, Some(buffer)) => (buffer, &self.stencil_layouts),
671            _ => (&self.buffer, &self.layouts),
672        };
673        let data = self.retrieve_buffer(ctx, buffer, layouts).await;
674        buffer.unmap();
675        ReadbackData {
676            data,
677            layouts: layouts.clone(),
678        }
679    }
680
681    async fn is_zero(&self, ctx: &TestingContext, buffer: &Buffer, layouts: &[MipLayout]) -> bool {
682        let is_zero = self
683            .retrieve_buffer(ctx, buffer, layouts)
684            .await
685            .iter()
686            .all(|b| *b == 0);
687        buffer.unmap();
688        is_zero
689    }
690
691    pub async fn are_zero(&self, ctx: &TestingContext) -> bool {
692        let buffer_zero = self.is_zero(ctx, &self.buffer, &self.layouts).await;
693        let mut stencil_buffer_zero = true;
694        if let Some(buffer) = &self.buffer_stencil {
695            stencil_buffer_zero = self.is_zero(ctx, buffer, &self.stencil_layouts).await;
696        };
697        buffer_zero && stencil_buffer_zero
698    }
699
700    /// Assert that the color or depth aspect starts with `expected_data`.
701    ///
702    /// Only a prefix of the readback is checked, so for a texture with more than one mip
703    /// level this covers mip level 0 alone unless `expected_data` holds every mip level.
704    pub async fn assert_buffer_contents(&self, ctx: &TestingContext, expected_data: &[u8]) {
705        self.assert_buffer_contents_imprecise(ctx, expected_data, 0)
706            .await;
707    }
708
709    pub async fn assert_buffer_contents_imprecise(
710        &self,
711        ctx: &TestingContext,
712        expected_data: &[u8],
713        max_diff: u8,
714    ) {
715        let result_buffer = self.retrieve_buffer(ctx, &self.buffer, &self.layouts).await;
716        assert!(
717            result_buffer.len() >= expected_data.len(),
718            "Result buffer ({}) smaller than expected buffer ({})",
719            result_buffer.len(),
720            expected_data.len()
721        );
722        let result_buffer = &result_buffer[..expected_data.len()];
723        assert!(result_buffer
724            .iter()
725            .zip(expected_data)
726            .all(|(a, b)| a.abs_diff(*b) <= max_diff));
727        self.buffer.unmap();
728    }
729}

731/// Tightly packed contents of one aspect of a texture, as read back by
732/// [`ReadbackBuffers::retrieve`].
733pub struct ReadbackData {
734    data: Vec<u8>,
735    layouts: Vec<MipLayout>,
736}
737
738impl ReadbackData {
739    /// Every mip level, tightly packed, mip level 0 first.
740    pub fn all(&self) -> &[u8] {
741        &self.data
742    }
743
744    /// Contents of one array layer (for a `D2` texture) or depth slice (for `D3`) of one
745    /// mip level, tightly packed.
746    pub fn subresource(&self, mip_level: u32, index: u32) -> &[u8] {
747        let layouts = &self.layouts[..mip_level as usize + 1];
748        let (layout, preceding) = layouts.split_last().unwrap();
749        assert!(
750            index < layout.size.depth_or_array_layers,
751            "mip level {mip_level} has {} array layers or depth slices, asked for {index}",
752            layout.size.depth_or_array_layers,
753        );
754        let size = layout.unpadded_subresource_size();
755        let start = preceding
756            .iter()
757            .map(MipLayout::unpadded_size)
758            .sum::<usize>()
759            + index as usize * size;
760        &self.data[start..start + size]
761    }
