15/// This information determines if a test should run.
16#[derive(Clone)]
17pub struct TestParameters {
18    pub required_features: Features,
19    pub required_downlevel_caps: DownlevelCapabilities,
20    pub required_limits: Limits,
21
22    pub required_instance_flags: InstanceFlags,
23
24    /// Instance flags that must be cleared for this test, even if they would
25    /// otherwise be set by the harness defaults or the environment.
26    pub forbidden_instance_flags: InstanceFlags,
27
28    /// On Dx12, specifically test against the Fxc compiler.
29    ///
30    /// For testing workarounds to Fxc bugs.
31    pub force_fxc: bool,
32
33    /// Conditions under which this test should be skipped.
34    pub skips: Vec<FailureCase>,
35
36    /// Conditions under which this test should be run, but is expected to fail.
37    pub failures: Vec<FailureCase>,
38
39    /// For certain features (ray tracing), metal shader validation is completely
40    /// broken
41    pub disable_mtl_shader_validation: bool,
42}
43
44impl Default for TestParameters {
45    fn default() -> Self {
46        Self {
47            required_features: Features::empty(),
48            required_downlevel_caps: LOWEST_DOWNLEVEL_PROPERTIES,
49            required_limits: Limits::downlevel_webgl2_defaults(),
50            required_instance_flags: InstanceFlags::empty(),
51            forbidden_instance_flags: InstanceFlags::empty(),
52            force_fxc: false,
53            // By default we skip the noop backend, and enable it if the test
54            // parameters ask us to remove it.
55            skips: vec![FailureCase::backend(wgpu::Backends::NOOP)],
56            failures: Vec::new(),
57            disable_mtl_shader_validation: false,
58        }
59    }
60}

62// Builder pattern to make it easier
63impl TestParameters {
64    /// Set of common features that most internal tests require for compute and readback.
65    pub fn test_features_limits(self) -> Self {
66        self.downlevel_flags(DownlevelFlags::COMPUTE_SHADERS)
67            .limits(wgpu::Limits::downlevel_defaults())
68    }
69
70    /// Set the list of features this test requires.
71    pub fn features(mut self, features: Features) -> Self {
72        self.required_features |= features;
73        self
74    }
75
76    pub fn downlevel_flags(mut self, downlevel_flags: DownlevelFlags) -> Self {
77        self.required_downlevel_caps.flags |= downlevel_flags;
78        self
79    }
80
81    /// Set the limits needed for the test.
82    pub fn limits(mut self, limits: Limits) -> Self {
83        self.required_limits = limits;
84        self
85    }
86
87    /// Sets the instance flags that the test requires.
88    pub fn instance_flags(mut self, instance_flags: InstanceFlags) -> Self {
89        self.required_instance_flags |= instance_flags;
90        self
91    }
92
93    /// Removes instance flags that are incompatible with the current test,
94    /// overriding the harness defaults and the environment.
95    pub fn remove_instance_flags(mut self, instance_flags: InstanceFlags) -> Self {
96        self.forbidden_instance_flags |= instance_flags;
97        self
98    }
99
100    pub fn force_fxc(mut self, force_fxc: bool) -> Self {
101        self.force_fxc = force_fxc;
102        self
103    }
104
105    /// Mark the test as always failing, but not to be skipped.
106    pub fn expect_fail(mut self, when: FailureCase) -> Self {
107        self.failures.push(when);
108        self
109    }
110
111    /// Mark the test as always failing, and needing to be skipped.
112    pub fn skip(mut self, when: FailureCase) -> Self {
113        self.skips.push(when);
114        self
115    }
116
117    /// Enable testing against the noop backend and miri.
118    ///
119    /// The noop backend does not execute any operations, but allows us to test
120    /// validation and memory safety.
121    pub fn enable_noop(mut self) -> Self {
122        self.skips
123            .retain(|case| *case != FailureCase::backend(wgpu::Backends::NOOP));
124        self
125    }
126
127    /// Disable metal shader validation.
128    ///
129    /// Metal shader validation can cause features (ray tracing specifically)
130    /// to break. This disables it so it can be tested.
131    pub fn disable_mtl_shader_validation(mut self) -> Self {
132        self.disable_mtl_shader_validation = true;
133        self
134    }
135}

137/// Information about a test, including if if it should be skipped.
138pub struct TestInfo {
139    pub skip: bool,
140    pub failure_application_reasons: FailureApplicationReasons,
141    pub failures: Vec<FailureCase>,
142    pub running_msg: String,
143}
144
145impl TestInfo {
146    pub(crate) fn from_configuration(test: &GpuTestConfiguration, adapter: &AdapterReport) -> Self {
147        // Figure out if a test is unsupported, and why.
148        let mut unsupported_reasons: ArrayVec<_, 4> = ArrayVec::new();
149        let missing_features = test.params.required_features - adapter.features;
150        if !missing_features.is_empty() {
151            unsupported_reasons.push("Features");
152        }
153
154        if !test.params.required_limits.check_limits(&adapter.limits) {
155            unsupported_reasons.push("Limits");
156        }
157
158        let missing_downlevel_flags =
159            test.params.required_downlevel_caps.flags - adapter.downlevel_caps.flags;
160        if !missing_downlevel_flags.is_empty() {
161            unsupported_reasons.push("Downlevel Flags");
162        }
163
164        if test.params.required_downlevel_caps.shader_model > adapter.downlevel_caps.shader_model {
165            unsupported_reasons.push("Shader Model");
166        }
167
168        // Produce a lower-case version of the adapter info, for comparison against
169        // `parameters.skips` and `parameters.failures`.
170        let adapter_lowercase_info = wgpu::AdapterInfo {
171            name: adapter.info.name.to_lowercase(),
172            driver: adapter.info.driver.to_lowercase(),
173            ..adapter.info.clone()
174        };
175
176        // Check if we should skip the test altogether.
177        let skip_application_reason = test
178            .params
179            .skips
180            .iter()
181            .find_map(|case| case.applies_to_adapter(&adapter_lowercase_info));
182
183        let mut applicable_cases = Vec::with_capacity(test.params.failures.len());
184        let mut failure_application_reasons = FailureApplicationReasons::empty();
185        let mut flaky = false;
186        for failure in &test.params.failures {
187            if let Some(reasons) = failure.applies_to_adapter(&adapter_lowercase_info) {
188                failure_application_reasons.insert(reasons);
189                applicable_cases.push(failure.clone());
190                flaky |= matches!(failure.behavior, FailureBehavior::Ignore);
191            }
192        }
193
194        let mut skip = false;
195        let running_msg = if let Some(reasons) = skip_application_reason {
196            skip = true;
197
198            let names: ArrayVec<_, 4> = reasons.iter_names().map(|(name, _)| name).collect();
199            let names_text = names.join(" | ");
200
201            format!("Skipped Failure: {names_text}")
202        } else if !unsupported_reasons.is_empty() {
203            skip = true;
204            format!("Unsupported: {}", unsupported_reasons.join(" | "))
205        } else if !failure_application_reasons.is_empty() {
206            if cfg!(target_arch = "wasm32") {
207                skip = true;
208            }
209
210            let names: ArrayVec<_, 4> = failure_application_reasons
211                .iter_names()
212                .map(|(name, _)| name)
213                .collect();
214            let names_text = names.join(" & ");
215            let flaky_text = if flaky { " Flaky " } else { " " };
216
217            format!("Executed{flaky_text}Failure: {names_text}")
218        } else {
219            String::from("Executed")
220        };
221
222        Self {
223            skip,
224            failure_application_reasons,
225            failures: applicable_cases,
226            running_msg,
227        }
228    }
