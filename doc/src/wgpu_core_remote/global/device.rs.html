90    /// Assign `id_in` an error with the given `label`.
91    ///
92    /// Ensure that future attempts to use `id_in` as a buffer ID will propagate
93    /// the error, following the WebGPU ["contagious invalidity"] style.
94    ///
95    /// Firefox uses this function to comply strictly with the WebGPU spec,
96    /// which requires [`GPUBufferDescriptor`] validation to be generated on the
97    /// Device timeline and leave the newly created [`GPUBuffer`] invalid.
98    ///
99    /// Ideally, we would simply let [`Device::create_buffer`] take care of all
100    /// of this, but some errors must be detected before we can even construct a
101    /// [`wgpu_types::BufferDescriptor`] to give it. For example, the WebGPU API
102    /// allows a `GPUBufferDescriptor`'s [`usage`] property to be any WebIDL
103    /// `unsigned long` value, but we can't construct a
104    /// [`wgpu_types::BufferUsages`] value from values with unassigned bits
105    /// set. This means we must validate `usage` before we can call
106    /// `Device::create_buffer`.
107    ///
108    /// When that validation fails, we must arrange for the buffer id to be
109    /// considered invalid. This method provides the means to do so.
110    ///
111    /// ["contagious invalidity"]: https://www.w3.org/TR/webgpu/#invalidity
112    /// [`GPUBufferDescriptor`]: https://www.w3.org/TR/webgpu/#dictdef-gpubufferdescriptor
113    /// [`GPUBuffer`]: https://www.w3.org/TR/webgpu/#gpubuffer
114    /// [`wgpu_types::BufferDescriptor`]: wgt::BufferDescriptor
115    /// [`Device::create_buffer`]: wgpu_core::device::Device::create_buffer
116    /// [`usage`]: https://www.w3.org/TR/webgpu/#dom-gputexturedescriptor-usage
117    /// [`wgpu_types::BufferUsages`]: wgt::BufferUsages
118    pub fn create_buffer_error(
119        &self,
120        device_id: DeviceId,
121        id_in: id::BufferId,
122        desc: &resource::BufferDescriptor,
123    ) {
124        let mut hub = self.hub.borrow_mut();
125        let Hub {
126            buffers, devices, ..
127        } = &mut *hub;
128        let device = devices.get(device_id);
129        buffers.assign(id_in, resource::Buffer::invalid(device, desc));
130    }

494    pub fn device_create_bind_group(
495        &self,
496        device_id: DeviceId,
497        desc: &BindGroupDescriptor,
498        id_in: id::BindGroupId,
499    ) {
500        let mut hub = self.hub.borrow_mut();
501        let Hub {
502            bind_groups,
503            devices,
504            bind_group_layouts,
505            buffers,
506            samplers,
507            texture_views,
508            external_textures,
509            ..
510        } = &mut *hub;
511
512        let device = devices.get(device_id);
513
514        let layout = bind_group_layouts.get(desc.layout);
515
516        fn resolve_entry<'a>(
517            e: &'a BindGroupEntry,
518            buffers: &mut Registry<Arc<resource::Buffer>>,
519            samplers: &mut Registry<Arc<resource::Sampler>>,
520            texture_views: &mut Registry<Arc<resource::TextureView>>,
521            external_textures: &mut Registry<Arc<resource::ExternalTexture>>,
522        ) -> binding_model::BindGroupEntry<'a> {
523            let resolve_buffer = |bb: &BufferBinding| {
524                let buffer = buffers.get(bb.buffer);
525                binding_model::BufferBinding {
526                    buffer,
527                    offset: bb.offset,
528                    size: bb.size,
529                }
530            };
531            let resolve_sampler = |id: &id::SamplerId| samplers.get(*id);
532            let resolve_view = |id: &id::TextureViewId| texture_views.get(*id);
533            let resolve_external_texture = |id: &id::ExternalTextureId| external_textures.get(*id);
534            let resource = match e.resource {
535                BindingResource::Buffer(ref buffer) => {
536                    binding_model::BindingResource::Buffer(resolve_buffer(buffer))
537                }
538                BindingResource::Sampler(ref sampler) => {
539                    binding_model::BindingResource::Sampler(resolve_sampler(sampler))
540                }
541                BindingResource::TextureView(ref view) => {
542                    binding_model::BindingResource::TextureView(resolve_view(view))
543                }
544                BindingResource::ExternalTexture(ref et) => {
545                    binding_model::BindingResource::ExternalTexture(resolve_external_texture(et))
546                }
547            };
548            binding_model::BindGroupEntry {
549                binding: e.binding,
550                resource,
551            }
552        }
553
554        let entries = desc
555            .entries
556            .iter()
557            .map(|e| resolve_entry(e, buffers, samplers, texture_views, external_textures))
558            .collect::<Vec<_>>();
559        let entries = Cow::Owned(entries);
560
561        let desc = binding_model::BindGroupDescriptor {
562            label: desc.label.clone(),
563            layout,
564            entries,
565        };
566
567        let bind_group = device.create_bind_group(&desc);
568
569        bind_groups.assign(id_in, bind_group);
570    }

772    pub fn device_create_render_pipeline(
773        &self,
774        device_id: DeviceId,
775        desc: &RenderPipelineDescriptor,
776        id_in: id::RenderPipelineId,
777    ) -> (
778        id::RenderPipelineId,
779        Option<pipeline::CreateRenderPipelineError>,
780    ) {
781        let mut hub = self.hub.borrow_mut();
782        let Hub {
783            render_pipelines,
784            devices,
785            shader_modules,
786            pipeline_layouts,
787            pipeline_caches,
788            ..
789        } = &mut *hub;
790
791        let device = devices.get(device_id);
792
793        let layout = desc.layout.map(|layout| pipeline_layouts.get(layout));
794
795        let cache = desc.cache.map(|cache| pipeline_caches.get(cache));
796
797        let vertex = {
798            let module = shader_modules.get(desc.vertex.stage.module);
799            let stage = ProgrammableStageDescriptor {
800                module,
801                entry_point: desc.vertex.stage.entry_point.clone(),
802                constants: desc.vertex.stage.constants.clone(),
803                zero_initialize_workgroup_memory: desc
804                    .vertex
805                    .stage
806                    .zero_initialize_workgroup_memory,
807            };
808            RenderPipelineVertexProcessor::Vertex(pipeline::VertexState {
809                stage,
810                buffers: desc.vertex.buffers.clone(),
811            })
812        };
813
814        let fragment = if let Some(ref state) = desc.fragment {
815            let module = shader_modules.get(state.stage.module);
816
817            let stage = ProgrammableStageDescriptor {
818                module,
819                entry_point: state.stage.entry_point.clone(),
820                constants: state.stage.constants.clone(),
821                zero_initialize_workgroup_memory: state.stage.zero_initialize_workgroup_memory,
822            };
823            Some(pipeline::FragmentState {
824                stage,
825                targets: state.targets.clone(),
826            })
827        } else {
828            None
829        };
830
831        let desc = ResolvedGeneralRenderPipelineDescriptor {
832            label: desc.label.clone(),
833            layout,
834            vertex,
835            primitive: desc.primitive,
836            depth_stencil: desc.depth_stencil.clone(),
837            multisample: desc.multisample,
838            fragment,
839            multiview_mask: desc.multiview_mask,
840            cache,
841        };
842
843        let (pipeline, error) = device.create_render_pipeline(desc);
844
845        let id = render_pipelines.assign(id_in, pipeline);
846
847        (id, error)
848    }
