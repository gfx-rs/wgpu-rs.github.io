43pub trait Trace: Any + Send + Sync {
44    fn make_binary(&mut self, kind: DataKind, data: &[u8]) -> Data;
45
46    fn make_string(&mut self, kind: DataKind, data: &str) -> Data;
47
48    fn add(&mut self, action: Action<'_, PointerReferences>)
49    where

53#[derive(Debug)]
54pub struct DiskTrace {
55    path: std::path::PathBuf,
56    file: std::fs::File,
57    config: ron::ser::PrettyConfig,
58    data_id: usize,
59}
60
61impl DiskTrace {
62    pub fn new(path: std::path::PathBuf) -> Result<Self, std::io::Error> {
63        log::debug!("Tracing into '{path:?}'");
64        let mut file = std::fs::File::create(path.join(FILE_NAME))?;
65        file.write_all(b"[\n")?;
66        Ok(Self {
67            path,
68            file,
69            config: ron::ser::PrettyConfig::default(),
70            data_id: 0,
71        })
72    }
73}

75impl Trace for DiskTrace {
76    /// Store `[u8]` data in the trace.
77    ///
78    /// Using a string `kind` is probably a bug, but should work as long as the
79    /// data is UTF-8.
80    fn make_binary(&mut self, kind: DataKind, data: &[u8]) -> Data {
81        self.data_id += 1;
82        let name = std::format!("data{}.{}", self.data_id, kind);
83        let _ = std::fs::write(self.path.join(&name), data);
84        Data::File(name)
85    }
86
87    /// Store `str` data in the trace.
88    ///
89    /// Using a binary `kind` is fine, but it's not clear why not use
90    /// `make_binary` instead.
91    fn make_string(&mut self, kind: DataKind, data: &str) -> Data {
92        self.make_binary(kind, data.as_bytes())
93    }
94
95    fn add(&mut self, action: Action<'_, PointerReferences>)
96    where
97        for<'a> Action<'a, PointerReferences>: serde::Serialize,
98    {
99        match ron::ser::to_string_pretty(&action, self.config.clone()) {
100            Ok(string) => {
101                let _ = writeln!(self.file, "{string},");
102            }
103            Err(e) => {
104                log::warn!("RON serialization failure: {e:?}");
105            }
106        }
107    }
108}
109
110impl Drop for DiskTrace {
111    fn drop(&mut self) {
112        let _ = self.file.write_all(b"]");
113    }
114}

714                    .iter()
715                    .map(|entry| {
716                        let resource = match &entry.resource {
717                            BindingResource::Buffer(buffer_binding) => {
718                                BindingResource::Buffer(BufferBinding {
719                                    buffer: buffer_binding.buffer.to_trace(),
720                                    offset: buffer_binding.offset,
721                                    size: buffer_binding.size,
722                                })
723                            }
724                            BindingResource::BufferArray(buffer_bindings) => {
725                                let resolved_buffers: Vec<_> = buffer_bindings
726                                    .iter()
727                                    .map(|bb| BufferBinding {
728                                        buffer: bb.buffer.to_trace(),
729                                        offset: bb.offset,
730                                        size: bb.size,
731                                    })
732                                    .collect();
733                                BindingResource::BufferArray(Cow::Owned(resolved_buffers))
734                            }
735                            BindingResource::Sampler(sampler_id) => {
736                                BindingResource::Sampler(sampler_id.to_trace())
737                            }
738                            BindingResource::SamplerArray(sampler_ids) => {
739                                let resolved: Vec<_> =
740                                    sampler_ids.iter().map(|id| id.to_trace()).collect();
741                                BindingResource::SamplerArray(Cow::Owned(resolved))
742                            }
743                            BindingResource::TextureView(texture_view_id) => {
744                                BindingResource::TextureView(texture_view_id.to_trace())
745                            }
746                            BindingResource::TextureViewArray(texture_view_ids) => {
747                                let resolved: Vec<_> =
748                                    texture_view_ids.iter().map(|id| id.to_trace()).collect();
749                                BindingResource::TextureViewArray(Cow::Owned(resolved))
750                            }
751                            BindingResource::AccelerationStructure(tlas_id) => {
752                                BindingResource::AccelerationStructure(tlas_id.to_trace())
753                            }
754                            BindingResource::AccelerationStructureArray(tlas_ids) => {
755                                let resolved: Vec<_> =
756                                    tlas_ids.iter().map(|id| id.to_trace()).collect();
757                                BindingResource::AccelerationStructureArray(Cow::Owned(resolved))
758                            }
759                            BindingResource::ExternalTexture(external_texture_id) => {
760                                BindingResource::ExternalTexture(external_texture_id.to_trace())
761                            }
762                        };
763                        BindGroupEntry {
764                            binding: entry.binding,
765                            resource,
766                        }
767                    })
768                    .collect(),
769            ),
770        }
771    }
772}
