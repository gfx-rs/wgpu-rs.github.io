1//! This crate contains types for the remoting version of wgpu-core for browser implementing WebGPU.
2//!
3//! It contains types that are used for IPC communication between the browser's untrusted content process
4//! and the browser's trusted GPU process and [`IdentityHub`](crate::identity::IdentityHub) for Id generation.
5//! All in all it contains all types needed for content process.
6//!
7//! All IPC types implement `serde::Serialize` and `serde::Deserialize` so that they can be sent over IPC.
8//!
9//! Types are defined entirely separately from wgpu-core's (and eventually wgpu-types's) public types,
10//! so that even as experimental features like raytracing and native wpgu features that are not standard WebGPU,
11//! cannot be expressed in untrusted (thus potentially malicious) content processes,
12//! as Serde's standard deserialization rejects invalid values of the IPC types.
13//!
14//! For more information about the remoting architecture, see the wgpu-core-remote crate's documentation.

35/// Options for requesting adapter.
36///
37/// Corresponds to [WebGPU `GPURequestAdapterOptions`](
38/// https://gpuweb.github.io/gpuweb/#dictdef-gpurequestadapteroptions).
39#[repr(C)]
40#[derive(Clone, Debug, PartialEq, Eq, Hash, serde::Serialize, serde::Deserialize)]
41pub struct RequestAdapterOptions {
42    /// Power preference for the adapter.
43    pub power_preference: wgt::PowerPreference,
44    /// Indicates that only a fallback adapter can be returned. This is generally a "software"
45    /// implementation on the system.
46    pub force_fallback_adapter: bool,
47}
48
49assert_ffi_safe!(RequestAdapterOptions);

51/// Describes a `Device`.
52///
53/// Corresponds to [WebGPU `GPUDeviceDescriptor`](
54/// https://gpuweb.github.io/gpuweb/#gpudevicedescriptor).
55#[derive(Clone, Debug, Default, serde::Serialize, serde::Deserialize)]
56pub struct DeviceDescriptor<'a> {
57    /// Debug label for the device.
58    pub label: Label<'a>,
59    /// Specifies the features that are required by the device request.
60    /// The request will fail if the adapter cannot provide these features.
61    ///
62    /// Exactly the specified set of features, and no more or less,
63    /// will be allowed in validation of API calls on the resulting device.
64    pub required_features: wgt::FeaturesWebGPU,
65    /// Specifies the limits that are required by the device request.
66    /// The request will fail if the adapter cannot provide these limits.
67    ///
68    /// Exactly the specified limits, and no better or worse,
69    /// will be allowed in validation of API calls on the resulting device.
70    pub required_limits: wgt::Limits,
71    /// Specifies the descriptor of the default queue for the device request.
72    ///
73    /// Corresponds to [WebGPU `GPUDeviceDescriptor.defaultQueue`](https://gpuweb.github.io/gpuweb/#dom-gpudevicedescriptor-defaultqueue).
74    pub default_queue: QueueDescriptor<'a>,
75}

82/// Describes a `TextureView`.
83///
84/// Corresponds to [WebGPU `GPUTextureViewDescriptor`](https://gpuweb.github.io/gpuweb/#dictdef-gputextureviewdescriptor).
85#[derive(Clone, Debug, Default, Eq, PartialEq, serde::Serialize, serde::Deserialize)]
86#[serde(default)]
87pub struct TextureViewDescriptor<'a> {
88    /// Debug label of the texture view.
89    ///
90    /// This will show up in graphics debuggers for easy identification.
91    pub label: Label<'a>,
92    /// Format of the texture view, or `None` for the same format as the texture
93    /// itself.
94    ///
95    /// At this time, it must be the same the underlying format of the texture.
96    pub format: Option<wgt::TextureFormat>,
97    /// The dimension of the texture view.
98    ///
99    /// - For 1D textures, this must be `D1`.
100    /// - For 2D textures it must be one of `D2`, `D2Array`, `Cube`, or `CubeArray`.
101    /// - For 3D textures it must be `D3`.
102    pub dimension: Option<wgt::TextureViewDimension>,
103    /// The allowed usage(s) for the texture view. Must be a subset of the usage flags of the texture.
104    /// If not provided, defaults to the full set of usage flags of the texture.
105    pub usage: Option<wgt::TextureUsages>,
106    /// Range within the texture that is accessible via this view.
107    pub range: wgt::ImageSubresourceRange,
108    /// Texture component swizzle.
109    /// When the texture view is accessed by a shader, the red/green/blue/alpha channels are replaced
110    /// by the value corresponding to the component specified in [`wgt::TextureComponentSwizzle`].
111    ///
112    /// This requires [`wgt::Features::TEXTURE_COMPONENT_SWIZZLE`] if it is not identity swizzle.
113    pub swizzle: wgt::TextureComponentSwizzle,
114}

116/// Describes a `Sampler`
117///
118/// Corresponds to [WebGPU `GPUSamplerDescriptor`](https://gpuweb.github.io/gpuweb/#dictdef-gpusamplerdescriptor).
119#[derive(Clone, Debug, PartialEq, serde::Serialize, serde::Deserialize)]
120pub struct SamplerDescriptor<'a> {
121    /// Debug label of the sampler.
122    ///
123    /// This will show up in graphics debuggers for easy identification.
124    pub label: Label<'a>,
125    /// How to deal with out of bounds accesses in the u (i.e. x) direction
126    pub address_modes: [wgt::AddressMode; 3],
127    /// How to filter the texture when it needs to be magnified (made larger)
128    pub mag_filter: wgt::FilterMode,
129    /// How to filter the texture when it needs to be minified (made smaller)
130    pub min_filter: wgt::FilterMode,
131    /// How to filter between mip map levels
132    pub mipmap_filter: wgt::MipmapFilterMode,
133    /// Minimum level of detail (i.e. mip level) to use
134    pub lod_min_clamp: f32,
135    /// Maximum level of detail (i.e. mip level) to use
136    pub lod_max_clamp: f32,
137    /// If this is enabled, this is a comparison sampler using the given comparison function.
138    pub compare: Option<wgt::CompareFunction>,
139    /// Must be at least 1. If this is not 1, all filter modes must be linear.
140    pub anisotropy_clamp: u16,
141}

143/// Describes a pipeline layout.
144///
145/// A `PipelineLayoutDescriptor` can be used to create a pipeline layout.
146#[derive(Clone, Debug, PartialEq, Eq, Hash, serde::Serialize, serde::Deserialize)]
147/// cbindgen:ignore
148pub struct PipelineLayoutDescriptor<'a> {
149    /// Debug label of the pipeline layout.
150    ///
151    /// This will show up in graphics debuggers for easy identification.
152    pub label: Label<'a>,
153    /// Bind groups that this pipeline uses. The first entry will provide all the bindings for
154    /// "set = 0", second entry will provide all the bindings for "set = 1" etc.
155    pub bind_group_layouts: Cow<'a, [Option<id::BindGroupLayoutId>]>,
156    /// The number of bytes of immediate data that are allocated for use
157    /// in the shader. The `var<immediate>`s in the shader attached to
158    /// this pipeline must be equal or smaller than this size.
159    ///
160    /// If this value is non-zero, [`wgt::Features::IMMEDIATES`] must be enabled.
161    pub immediate_size: u32,
162}
