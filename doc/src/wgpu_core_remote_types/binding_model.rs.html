8/// Corresponds to [`GPUBufferBinding`](https://www.w3.org/TR/webgpu/#dictdef-gpubufferbinding).
9#[repr(C)]
10#[derive(Clone, Debug, Hash, Eq, PartialEq, Serialize, Deserialize)]
11pub struct BufferBinding {
12    pub buffer: BufferId,
13    pub offset: wgt::BufferAddress,
14
15    /// Size of the binding. If `None`, the binding spans from `offset` to the
16    /// end of the buffer.
17    ///
18    /// We use `BufferAddress` to allow a size of zero on this `wgpu_core` type,
19    /// because JavaScript bindings cannot readily express `Option<NonZeroU64>`.
20    /// The `wgpu` API uses `Option<BufferSize>` (i.e. `NonZeroU64`) for this
21    /// field.
22    pub size: Option<wgt::BufferAddress>,
23}

25/// Corresponds to [`GPUBindingResource`](https://www.w3.org/TR/webgpu/#typedefdef-gpubindingresource).

34/// Bindable resource and the slot to bind it to.
35///
36/// This corresponds to [`GPUBindGroupEntry`](https://www.w3.org/TR/webgpu/#dictdef-gpubindgroupentry).
37#[derive(Clone, Debug, Serialize, Deserialize)]
38pub struct BindGroupEntry {
39    /// Slot for which binding provides resource. Corresponds to an entry of the same
40    /// binding index in the [`BindGroupLayoutDescriptor`].
41    pub binding: u32,
42    /// Resource to attach to the binding
43    pub resource: BindingResource,
44}
45
46/// Describes a group of bindings and the resources to be bound.
47///
48/// This corresponds to [`GPUBindGroupDescriptor`](https://www.w3.org/TR/webgpu/#dictdef-gpubindgroupdescriptor).
49#[derive(Clone, Debug, Serialize, Deserialize)]
50pub struct BindGroupDescriptor<'a> {
51    /// Debug label of the bind group.
52    ///
53    /// This will show up in graphics debuggers for easy identification.
54    pub label: Label<'a>,
55    /// The BindGroupLayout that corresponds to this bind group.
56    pub layout: BindGroupLayoutId,
57    /// The resources to bind to this bind group.
58    pub entries: Cow<'a, [BindGroupEntry]>,
59}

61/// Corresponds to [`GPUBindGroupLayoutDescriptor`](https://www.w3.org/TR/webgpu/#dictdef-gpubindgrouplayoutdescriptor).
62#[derive(Clone, Debug, Serialize, Deserialize)]
63pub struct BindGroupLayoutDescriptor<'a> {
64    /// Debug label of the bind group layout.
65    ///
66    /// This will show up in graphics debuggers for easy identification.
67    pub label: Label<'a>,
68    /// Array of entries in this BindGroupLayout
69    pub entries: Cow<'a, [wgt::BindGroupLayoutEntry]>,
