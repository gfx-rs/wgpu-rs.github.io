9/// Corresponds to [`GPUBufferBinding`](https://www.w3.org/TR/webgpu/#dictdef-gpubufferbinding).
10#[repr(C)]
11#[derive(Clone, Debug, Hash, Eq, PartialEq, Serialize, Deserialize)]
12pub struct BufferBinding {
13    pub buffer: BufferId,
14    pub offset: wgt::BufferAddress,
15
16    /// Size of the binding. If `None`, the binding spans from `offset` to the
17    /// end of the buffer.
18    ///
19    /// We use `BufferAddress` to allow a size of zero on this `wgpu_core` type,
20    /// because JavaScript bindings cannot readily express `Option<NonZeroU64>`.
21    /// The `wgpu` API uses `Option<BufferSize>` (i.e. `NonZeroU64`) for this
22    /// field.
23    pub size: FfiOption<wgt::BufferAddress>,
24}
25
26assert_ffi_safe!(BufferBinding);

40/// Bindable resource and the slot to bind it to.
41///
42/// This corresponds to [`GPUBindGroupEntry`](https://www.w3.org/TR/webgpu/#dictdef-gpubindgroupentry).
43#[repr(C)]
44#[derive(Clone, Debug, Serialize, Deserialize)]
45pub struct BindGroupEntry {
46    /// Slot for which binding provides resource. Corresponds to an entry of the same
47    /// binding index in the [`BindGroupLayoutDescriptor`].
48    pub binding: u32,
49    /// Resource to attach to the binding
50    pub resource: BindingResource,
51}
52
53assert_ffi_safe!(BindGroupEntry);
54
55/// Describes a group of bindings and the resources to be bound.
56///
57/// This corresponds to [`GPUBindGroupDescriptor`](https://www.w3.org/TR/webgpu/#dictdef-gpubindgroupdescriptor).
58#[derive(Clone, Debug, Serialize, Deserialize)]
59pub struct BindGroupDescriptor<'a> {
60    /// Debug label of the bind group.
61    ///
62    /// This will show up in graphics debuggers for easy identification.
63    pub label: Label<'a>,
64    /// The BindGroupLayout that corresponds to this bind group.
65    pub layout: BindGroupLayoutId,
66    /// The resources to bind to this bind group.
67    pub entries: Cow<'a, [BindGroupEntry]>,
68}

70/// Corresponds to [`GPUBindGroupLayoutDescriptor`](https://www.w3.org/TR/webgpu/#dictdef-gpubindgrouplayoutdescriptor).
71#[derive(Clone, Debug, Serialize, Deserialize)]
72pub struct BindGroupLayoutDescriptor<'a> {
73    /// Debug label of the bind group layout.
74    ///
75    /// This will show up in graphics debuggers for easy identification.
76    pub label: Label<'a>,
77    /// Array of entries in this BindGroupLayout
78    pub entries: Cow<'a, [wgt::BindGroupLayoutEntry]>,
79}
