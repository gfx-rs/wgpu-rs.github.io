33    },
34    CopyTextureToTexture {
35        source: wgt::TexelCopyTextureInfo<id::TextureId>,
36        destination: wgt::TexelCopyTextureInfo<id::TextureId>,
37        copy_size: wgt::Extent3d,
38    },
39    ClearBuffer {
40        buffer: id::BufferId,
41        offset: u64, // optional, defaults to 0
42        size: Option<u64>,
43    },
44    ResolveQuerySet {
45        query_set: id::QuerySetId,
46        first_query: u32,
47        query_count: u32,
48        destination: id::BufferId,

52    Finish {
53        desc: CommandBufferDescriptor<'a>,
