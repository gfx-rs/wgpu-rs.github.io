16/// Backends supported by wgpu.
17///
18/// See also [`Backends`].
19#[repr(u8)]
20#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
21#[cfg_attr(feature = "serde", derive(serde::Serialize, serde::Deserialize))]
22pub enum Backend {
23    /// Dummy backend, which may be used for testing.
24    ///
25    /// It performs no rendering or computation, but allows creation of stub GPU resource types,
26    /// so that code which manages GPU resources can be tested without an available GPU.
27    /// Specifically, the following operations are implemented:
28    ///
29    /// * Enumerating adapters will always return one noop adapter, which can be used to create
30    ///   devices.
31    /// * Buffers may be created, written, mapped, and copied to other buffers.
32    /// * Command encoders may be created, but only buffer operations are useful.
33    ///
34    /// Other resources can be created but are nonfunctional; notably,
35    ///
36    /// * Render passes and compute passes are not executed.
37    /// * Textures may be created, but do not store any texels.
38    /// * There are no compatible surfaces.
39    ///
40    /// An adapter using the noop backend can only be obtained if [`NoopBackendOptions`]
41    /// enables it, in addition to the ordinary requirement of [`Backends::NOOP`] being set.
42    /// This ensures that applications not desiring a non-functional backend will not receive it.
43    Noop = 0,
44
45    /// Vulkan API (Windows, Linux, Android, MacOS via `vulkan-portability`/MoltenVK)
46    Vulkan = 1,
47
48    /// Metal API (Apple platforms)
49    Metal = 2,
50
51    /// Direct3D-12 (Windows)
52    Dx12 = 3,
53
54    /// OpenGL 3.3+, OpenGL ES 3.0+, WebGL2.
55    ///
56    /// - On Windows, this is normally OpenGL. If you build with
57    ///   `cfg(windows_angle)` to use ANGLE instead, then this is OpenGL ES.
58    ///
59    /// - On Linux, we create a full OpenGL context if possible (Mesa offers
60    ///   it), but if that fails we fall back to OpenGL ES.
61    ///
62    /// - On Android and macOS with ANGLE, we get OpenGL ES.
63    ///
64    /// - When running in a web browser, we get WebGL2.
65    Gl = 4,
66
67    /// WebGPU in the browser
68    BrowserWebGpu = 5,
69}
70
71impl Backend {
72    /// Array of all [`Backend`] values, corresponding to [`Backends::all()`].
73    pub const ALL: [Backend; Backends::all().bits().count_ones() as usize] = [
74        Self::Noop,
75        Self::Vulkan,
76        Self::Metal,
77        Self::Dx12,
78        Self::Gl,
79        Self::BrowserWebGpu,
80    ];
81
82    /// Returns the string name of the backend.
83    #[must_use]
84    pub const fn to_str(self) -> &'static str {
85        match self {
86            Backend::Noop => "noop",
87            Backend::Vulkan => "vulkan",
88            Backend::Metal => "metal",
89            Backend::Dx12 => "dx12",
90            Backend::Gl => "gl",
91            Backend::BrowserWebGpu => "webgpu",
92        }
93    }
94}

102bitflags::bitflags! {
103    /// Represents the backends that wgpu will use.
104    #[repr(transparent)]
105    #[cfg_attr(feature = "serde", derive(Serialize, Deserialize))]
106    #[cfg_attr(feature = "serde", serde(transparent))]
107    #[derive(Debug, Copy, Clone, PartialEq, Eq, Hash)]
108    pub struct Backends: u32 {
109        /// [`Backend::Noop`].
110        const NOOP = 1 << Backend::Noop as u32;
111
112        /// [`Backend::Vulkan`].
113        /// Supported on Windows, Linux/Android, and macOS/iOS via Vulkan Portability (with the Vulkan feature enabled)
114        const VULKAN = 1 << Backend::Vulkan as u32;
115
116        /// [`Backend::Gl`].
117        /// Supported on Linux/Android, the web through webassembly via WebGL, and
118        /// Windows through native OpenGL by default or ANGLE with `cfg(windows_angle)`.
119        const GL = 1 << Backend::Gl as u32;
120
121        /// [`Backend::Metal`].
122        /// Supported on macOS and iOS.
123        const METAL = 1 << Backend::Metal as u32;
124
125        /// [`Backend::Dx12`].
126        /// Supported on Windows 10 and later
127        const DX12 = 1 << Backend::Dx12 as u32;
128
129        /// [`Backend::BrowserWebGpu`].
130        /// Supported when targeting the web through WebAssembly with the `webgpu` feature enabled.
131        ///
132        /// The WebGPU backend is special in several ways:
133        /// It is not not implemented by `wgpu_core` and instead by the higher level `wgpu` crate.
134        /// Whether WebGPU is targeted is decided upon the creation of the `wgpu::Instance`,
135        /// *not* upon adapter creation. See `wgpu::Instance::new`.
136        const BROWSER_WEBGPU = 1 << Backend::BrowserWebGpu as u32;
137
138        /// All the apis that wgpu offers first tier of support for.
139        ///
140        /// * [`Backends::VULKAN`]
141        /// * [`Backends::METAL`]
142        /// * [`Backends::DX12`]
143        /// * [`Backends::BROWSER_WEBGPU`]
144        const PRIMARY = Self::VULKAN.bits()
145            | Self::METAL.bits()
146            | Self::DX12.bits()
147            | Self::BROWSER_WEBGPU.bits();
148
149        /// All the apis that wgpu offers second tier of support for. These may
150        /// be unsupported/still experimental.
151        ///
152        /// * [`Backends::GL`]
153        const SECONDARY = Self::GL.bits();
154    }
155}

169impl Backends {
170    /// Gets a set of backends from the environment variable `WGPU_BACKEND`.
171    ///
172    /// See [`Self::from_comma_list()`] for the format of the string.
173    pub fn from_env() -> Option<Self> {
174        let env = crate::env::var("WGPU_BACKEND")?;
175        Some(Self::from_comma_list(&env))
176    }
177
178    /// Takes the given options, modifies them based on the `WGPU_BACKEND` environment variable, and returns the result.
179    pub fn with_env(&self) -> Self {
180        if let Some(env) = Self::from_env() {
181            env
182        } else {
183            *self
184        }
185    }
186
187    /// Generates a set of backends from a comma separated list of case-insensitive backend names.
188    ///
189    /// Whitespace is stripped, so both 'gl, dx12' and 'gl,dx12' are valid.
190    ///
191    /// Always returns WEBGPU on wasm over webgpu.
192    ///
193    /// Names:
194    /// - vulkan = "vulkan" or "vk"
195    /// - dx12   = "dx12" or "d3d12"
196    /// - metal  = "metal" or "mtl"
197    /// - gles   = "opengl" or "gles" or "gl"
198    /// - webgpu = "webgpu"
199    pub fn from_comma_list(string: &str) -> Self {
200        let mut backends = Self::empty();
201        for backend in string.to_lowercase().split(',') {
202            backends |= match backend.trim() {
203                "vulkan" | "vk" => Self::VULKAN,
204                "dx12" | "d3d12" => Self::DX12,
205                "metal" | "mtl" => Self::METAL,
206                "opengl" | "gles" | "gl" => Self::GL,
207                "webgpu" => Self::BROWSER_WEBGPU,
208                "noop" => Self::NOOP,
209                b => {
210                    log::warn!("unknown backend string '{b}'");
211                    continue;
212                }
213            }
214        }
215
216        if backends.is_empty() {
217            log::warn!("no valid backend strings found!");
218        }
219
220        backends
221    }
222}

323/// Controls whether OpenGL debug functions are enabled.
324///
325/// Debug functions include `glPushDebugGroup`, `glPopDebugGroup`, `glObjectLabel`, etc.
326/// These are useful for debugging but can cause crashes on some buggy drivers.
327#[derive(Clone, Copy, Debug, PartialEq, Eq, ConstDefault!)]
328pub enum GlDebugFns {
329    /// Automatically decide whether to enable debug functions.
330    ///
331    /// Debug functions will be enabled if supported by the driver, unless
332    /// running on a device known to have buggy debug function implementations
333    /// (e.g., Mali GPUs which can crash in `glPushDebugGroup`).
334    ///
335    /// This is the default behavior.
336    #[custom(default)]
337    Auto,
338    /// Force enable debug functions if supported by the driver.
339    ///
340    /// This ignores any device-specific workarounds and enables debug functions
341    /// on all devices that support them, including those with known bugs.
342    ForceEnabled,
343    /// Disable debug functions entirely.
344    ///
345    /// Debug functions will not be used even if supported by the driver.
346    Disabled,
347}
348
349impl GlDebugFns {
350    /// Choose debug functions setting from the environment variable `WGPU_GL_DEBUG_FNS`.
351    ///
352    /// Possible values (case insensitive):
353    /// - `auto` - automatically decide based on device
354    /// - `forceenabled`, `force_enabled`, or `enabled` - force enable
355    /// - `disabled` - disable entirely
356    ///
357    /// Use with `unwrap_or_default()` to get the default value if the environment variable is not set.
358    #[must_use]
359    pub fn from_env() -> Option<Self> {
360        let value = crate::env::var("WGPU_GL_DEBUG_FNS")
361            .as_deref()?
362            .to_lowercase();
363        match value.as_str() {
364            "auto" => Some(Self::Auto),
365            "forceenabled" | "force_enabled" | "enabled" => Some(Self::ForceEnabled),
366            "disabled" => Some(Self::Disabled),
367            _ => None,
368        }
369    }
370
371    /// Takes the given setting, modifies it based on the `WGPU_GL_DEBUG_FNS` environment variable, and returns the result.
372    ///
373    /// See `from_env` for more information.
374    #[must_use]
375    pub fn with_env(self) -> Self {
376        if let Some(debug_fns) = Self::from_env() {
377            debug_fns
378        } else {
379            self
380        }
381    }
382}

445        if let Some(v) = Self::from_env() {
446            v
447        } else {
448            self
449        }
450    }

453/// Configuration for loading a specific [DirectX 12 Agility SDK] runtime.
454///
455/// The Agility SDK allows applications to ship a newer version of the D3D12 runtime
456/// (`D3D12Core.dll`) alongside the application, enabling access to the latest D3D12
457/// features without waiting for the OS to update its built-in runtime. This is the
458/// standard way for games and applications to adopt new D3D12 functionality on older
459/// Windows versions.
460///
461/// Downloads and release notes are available at the [DirectX 12 Agility SDK] page.
462///
463/// wgpu loads the Agility SDK via the [Independent Devices API], which allows
464/// specifying the SDK path and version at runtime without requiring exported constants
465/// or developer mode. The [`sdk_version`](Self::sdk_version) must match the version of
466/// the `D3D12Core.dll` in the provided path exactly, or loading will fail, irrespective of
467/// the OS's built-in runtime version.
468///
469/// If the shipped SDK is older than the system runtime, the system runtime will be used.
470/// This allows applications to ship a minimum SDK version while still benefiting from SDK updates on the user's system.
471///
472/// If the Agility SDK fails to load (version mismatch, missing DLL, unsupported OS,
473/// etc.), the behavior is controlled by [`on_load_failure`](Self::on_load_failure).
474/// By default, wgpu logs a warning and falls back to the system-installed D3D12 runtime.
475/// Set it to [`Error`](Dx12AgilitySDKLoadFailure::Error) to fail instance creation instead
476/// (useful in CI/testing).
477///
478/// ## OS requirements
479///
480/// The Independent Devices API requires a Windows update from August/September 2023
481/// or newer:
482///
483/// - [Windows 11 21H2+ (KB5029332)][win11-21h2]
484/// - [Windows 10 22H2+ (KB5029331)][win10-22h2]
485/// - [Windows Server 2022+ (KB5030216)][server-2022]
486///
487/// On older OS builds the Agility SDK will not load and wgpu will log a warning and
488/// fall back to the system runtime (or error, per [`on_load_failure`](Self::on_load_failure)).
489///
490/// [DirectX 12 Agility SDK]: https://devblogs.microsoft.com/directx/directx12agility/
491/// [Independent Devices API]: https://devblogs.microsoft.com/directx/d3d12-independent-devices/
492/// [win11-21h2]: https://support.microsoft.com/en-us/topic/august-22-2023-kb5029332-os-build-22000-2360-preview-8f8aec64-77b4-4225-9a0f-f0153204ae28
493/// [win10-22h2]: https://support.microsoft.com/en-gb/topic/august-22-2023-kb5029331-os-build-19045-3393-preview-9f6c1dbd-0ee6-469b-af24-f9d0bf35ca18
494/// [server-2022]: https://support.microsoft.com/en-au/topic/september-12-2023-kb5030216-os-build-20348-1970-34d4aff3-fd05-4270-b288-4ab6379c7f81
495#[derive(Clone, Debug, Eq, PartialEq)]
496pub struct Dx12AgilitySDK {
497    /// The Agility SDK version number (e.g., 614 for SDK version 1.614.0).
498    ///
499    /// This must match the version of the `D3D12Core.dll` at [`sdk_path`](Self::sdk_path)
500    /// exactly, or the runtime will fail to load.
501    pub sdk_version: u32,
502    /// Path to the directory containing the Agility SDK's `D3D12Core.dll`.
503    pub sdk_path: String,
504    /// What to do if the Agility SDK fails to load.
505    ///
506    /// Defaults to [`Fallback`](Dx12AgilitySDKLoadFailure::Fallback).
507    ///
508    /// Can also be set via the `WGPU_DX12_AGILITY_SDK_REQUIRE` environment variable
509    /// (`1` for [`Error`](Dx12AgilitySDKLoadFailure::Error),
510    /// `0` for [`Fallback`](Dx12AgilitySDKLoadFailure::Fallback)).
511    pub on_load_failure: Dx12AgilitySDKLoadFailure,
512}
513
514impl Dx12AgilitySDK {
515    /// Read Agility SDK configuration from environment variables.
516    ///
517    /// Reads `WGPU_DX12_AGILITY_SDK_PATH`, `WGPU_DX12_AGILITY_SDK_VERSION`,
518    /// and `WGPU_DX12_AGILITY_SDK_REQUIRE`.
519    /// Both path and version must be set for this to return `Some`.
520    #[must_use]
521    pub fn from_env() -> Option<Self> {
522        let sdk_path = crate::env::var("WGPU_DX12_AGILITY_SDK_PATH")?;
523        let sdk_version_str = crate::env::var("WGPU_DX12_AGILITY_SDK_VERSION")?;
524        let sdk_version = sdk_version_str.parse::<u32>().ok()?;
525        let on_load_failure = Dx12AgilitySDKLoadFailure::from_env().unwrap_or_default();
526        Some(Self {
527            sdk_version,
528            sdk_path,
529            on_load_failure,
530        })
531    }
532
533    /// Takes the given configuration, overrides fields with environment variables if present,
534    /// and returns the result.
535    ///
536    /// Reads `WGPU_DX12_AGILITY_SDK_PATH`, `WGPU_DX12_AGILITY_SDK_VERSION`,
537    /// and `WGPU_DX12_AGILITY_SDK_REQUIRE`.
538    /// Each variable overrides the corresponding field independently.
539    #[must_use]
540    pub fn with_env(mut self) -> Self {
541        if let Some(sdk_path) = crate::env::var("WGPU_DX12_AGILITY_SDK_PATH") {
542            self.sdk_path = sdk_path;
543        }
544        if let Some(sdk_version_str) = crate::env::var("WGPU_DX12_AGILITY_SDK_VERSION") {
545            if let Ok(sdk_version) = sdk_version_str.parse::<u32>() {
546                self.sdk_version = sdk_version;
547            }
548        }
549        self.on_load_failure = self.on_load_failure.with_env();
550        self
551    }
552}

888impl DxcShaderModel {
889    /// Get the shader model supported by a certain DXC version.
890    pub fn from_dxc_version(major: u32, minor: u32) -> Self {

905        } else {
906            match minor {
907                0 => DxcShaderModel::V6_0,
908                1 => DxcShaderModel::V6_1,

914                7 => DxcShaderModel::V6_7,
915                8 => DxcShaderModel::V6_8,
916                9 => DxcShaderModel::V6_9,
